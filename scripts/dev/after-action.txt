# What a battle cost, on a page that stops for it.
#
#   STAHL_PRESENT=immediate STAHL_DEBUG=1 \
#     STAHL_SCRIPT=scripts/dev/after-action.txt cargo run -p stahlsenshamädchen
#
# No STAHL_BATTLE: this one starts on the campaign map, because the
# after-action report is a campaign screen and a scenario battle has no
# campaign to come back to.
#
# frontier puts Kuhlmann's 1st Company at offset [1, 1] -> axial (1, 1) and
# the Valkyrie Vanguard at [12, 1] -> axial (12, 1). Eleven hexes and four
# hexes of movement a day, so this marches east three days and runs into
# whatever the Valkyries have brought west.
#
# The three Enters that end the marching days are spaced by `wait` rather
# than `until idle`, which is the one place here a stopwatch is the right
# tool: they are keystrokes being spaced, not a simulation being waited on,
# and a campaign day settles in well under a second. Everything that is
# actually waiting on the game — the rounds of the battle, and the report at
# the end of it — uses `until`.

at 3.0
shot /tmp/aa-01-campaign.png

# Day one: select 1st Company and march it four hexes east.
hex 1,1
wait 0.5
click left
wait 0.5
hex 5,1
wait 0.5
click left
wait 1.0
key Enter
wait 4.0

# Day two, from where she stopped.
hex 5,1
wait 0.5
click left
wait 0.5
hex 9,1
wait 0.5
click left
wait 1.0
key Enter
wait 4.0

# Day three: into contact. If the Valkyries have already come west this walks
# straight onto them; if not, another day or two of Enters does it.
hex 9,1
wait 0.5
click left
wait 0.5
hex 12,1
wait 0.5
click left
wait 1.0
key Enter
wait 4.0
key Enter
wait 4.0
key Enter
wait 4.0

# From here one keystroke does double duty, which is why the tour is written
# this way rather than as two labelled halves: on the campaign map Enter ends
# the day, and once the two columns have run into each other the same key
# commits a round of the battle that started. So the tour presses Enter and
# waits for the game to settle, over and over, and does not need to know which
# screen it is on.
#
# There is deliberately no muster shot. Whether the prompt appears at all
# depends on whether anybody was near enough to join, and on this map on this
# day nobody was — the battle starts directly. A tour that insisted on
# photographing it would fail for a reason that is not a defect.

key Enter
until idle 90
key Enter
until idle 90
key Enter
until idle 90
key Enter
until idle 90
key Enter
until idle 90
key Enter
until idle 90
key Enter
until idle 90
key Enter
until idle 90
key Enter
until idle 90
key Enter
until idle 90
key Enter
until idle 90
key Enter
until idle 90
key Enter
until idle 90
key Enter
until idle 90
key Enter
until idle 90
key Enter
until idle 90
key Enter
until idle 90
key Enter
until idle 90
key Enter
until idle 90
key Enter
until idle 90

# ...and back to the campaign, where the bill is waiting. This is the shot
# this tour exists for: named cadets, what happened to each, and a campaign
# that has stopped to say so instead of narrating it past in among the tile
# income at three lines a second.
until waiting 180
expect log "Battle"
shot /tmp/aa-04-after-action.png

# One key dismisses it and the campaign runs on.
key Enter
until idle 30
shot /tmp/aa-05-dismissed.png

quit
